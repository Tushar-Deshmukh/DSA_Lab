q: "`strs` is a list of binary strings. Return the size of the largest subset of `strs` with at most `m` zeros and `n` ones in total. Example: `[\"10\",\"0001\",\"111001\",\"1\",\"0\"]`, m = 5, n = 3 → 4."

starter: |
  function findMaxForm(strs, m, n) {

  }
solution: |
  function findMaxForm(strs, m, n) {
    const dp = Array.from({ length: m + 1 }, () => new Array(n + 1).fill(0));
    for (const s of strs) {

      for (let i = m; i >= z; i--) for (let j = n; j >= o; j--) dp[i][j] = Math.max(dp[i][j], dp[i - z][j - o] + 1);

starter: |
  function minDeletions(a, b) {

  }
solution: |
  function minDeletions(a, b) {
    const m = a.length, n = b.length;
    let prev = new Array(n + 1).fill(0);
    for (let i = 1; i <= m; i++) {
      const cur = new Array(n + 1).fill(0);

  }

  brute: "Try flip sets: exponential and unnecessary; bits are independent."
  optimize: |

      flips += z ? (x | y ? 0 : 1) : x + y;
      a >>>= 1; b >>>= 1; c >>>= 1;
