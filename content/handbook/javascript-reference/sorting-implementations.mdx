Merge sort is in the Module 3 theory, Example 4. Quicksort with a random pivot and three-way partitioning (handles many duplicates, which a two-way partition does badly):

```js
function swap(a, i, j) { const t = a[i]; a[i] = a[j]; a[j] = t; }

function quickSort(a, lo = 0, hi = a.length - 1) {
  if (lo >= hi) return a;
  const pivot = a[lo + Math.floor(Math.random() * (hi - lo + 1))];
  let lt = lo, i = lo, gt = hi;
  // invariant: a[lo..lt-1] < pivot, a[lt..i-1] === pivot,
  //            a[gt+1..hi] > pivot
  while (i <= gt) {
    if (a[i] < pivot) swap(a, lt++, i++);
    // do not advance i: the swapped-in value is unchecked
    else if (a[i] > pivot) swap(a, i, gt--);
    else i++;
  }
  quickSort(a, lo, lt - 1);
  quickSort(a, gt + 1, hi);
  return a;
}
```

Dry run in Module 3 lesson 5: `[3, 1, 3, 2, 3]` with pivot 3 ends as `[1, 2, 3, 3, 3]` after one partition, with lt = 2 and gt = 4.
